cmake_minimum_required(VERSION 3.22)
project(omabridge_jni LANGUAGES CXX)
if(NOT DEFINED BRIDGE_JNI_SYMBOL OR NOT BRIDGE_JNI_SYMBOL MATCHES "^Java_[A-Za-z0-9_]+$")
 message(FATAL_ERROR "Set BRIDGE_JNI_SYMBOL to the consuming application's JNI entry point")
endif()
add_library(omabridge_jni SHARED bridge_jni.cpp)
target_compile_definitions(omabridge_jni PRIVATE OMABRIDGE_JNI_CALL=${BRIDGE_JNI_SYMBOL})
if(NOT ANDROID)
 find_package(JNI REQUIRED)
 target_include_directories(omabridge_jni PRIVATE ${JNI_INCLUDE_DIRS})
endif()
target_link_libraries(omabridge_jni PRIVATE "${BRIDGE_ARCHIVE}")
if(ANDROID)
 target_link_libraries(omabridge_jni PRIVATE log dl m)
 target_link_options(omabridge_jni PRIVATE "-Wl,-z,max-page-size=16384")
else()
 target_link_libraries(omabridge_jni PRIVATE pthread dl m)
endif()

target_link_options(omabridge_jni PRIVATE "-Wl,--exclude-libs,ALL")
